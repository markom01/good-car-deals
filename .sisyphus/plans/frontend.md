## Context

### Original Request
Create a minimal MVP frontend (Next.js + TypeScript + shadcn/ui) that displays a list of car deal listings from the NestJS backend's `/deals` API. Reuse backend types, minimize custom code, and use a theme file. Use specific Radix components (Badge, Button, Separator, ScrollArea) with shadcn/ui's default badge variants for deal classification.

### Interview Summary
**Key Discussions**:
- **Type sharing**: Both frontend and backend import the `Listing` interface from `@good-car-deals/shared` — a shared npm workspace package. The shared interface mirrors the Prisma model's serialized shape (BigInt → number, DateTime → string). Backend continues using Prisma's generated `Listing` type internally; the shared type represents the wire format contract. Frontend uses shared type exclusively — zero duplication.
- **Backend URL**: `http://localhost:3001/deals` (backend runs on port 3001)
- **State management**: `@tanstack/react-query` (React Query) for all API interactions
- **Display fields**: title, price, model, mileage, year, dealScore, pricePerYear
- **Badge colors**: shadcn/ui default variants (default, secondary, destructive)
- **Testing**: None for MVP — all verification via agent-executed QA (curl, Playwright/screenshots)
- **CORS**: Backend needs `app.enableCors()` — this is a backend change but critical for frontend to work

**Metis Findings** (incorporated):
- Backend port defaults to 3000 (`process.env.PORT ?? 3000`), not 3001 — need to set PORT=3001 in backend `.env` or change frontend target
- No CORS on backend — MUST add `app.enableCors()` or use Next.js rewrites
- `dealType` is null in 45/48 records — badge needs to handle null gracefully
- `dealScore` and `pricePerYear` are Floats — need `toFixed(2)` formatting
- `id` is Prisma BigInt, serialized as number via JSON — type interface should use `number`
- shadcn preset decodes to: green theme, taupe base, luma style, small radius, Inter/Roboto fonts, Lucide icons

  **What to do**:

  - Create `good-car-deals-frontend/src/lib/providers.tsx` — React Query provider wrapping the app:
    ```typescript
    'use client';

    import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
    import { useState, type ReactNode } from 'react';

    export function Providers({ children }: { children: ReactNode }) {
      const [queryClient] = useState(
        () =>
          new QueryClient({
            defaultOptions: {
              queries: {
                staleTime: 5 * 60 * 1000, // 5 min — data doesn't change often
                retry: 2,
                refetchOnWindowFocus: false,
              },
            },
          })
      );

      return (
        <QueryClientProvider client={queryClient}>
          {children}
        </QueryClientProvider>
      );
    }
    ```
  - Update `good-car-deals-frontend/src/app/layout.tsx` to wrap children with `<Providers>`
  - Create `good-car-deals-frontend/src/hooks/use-deals.ts` — React Query hook:
    ```typescript
    import { useQuery } from '@tanstack/react-query';
    import type { Listing } from '@good-car-deals/shared';
    
    const API_URL = process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001';
    
    async function fetchDeals(): Promise<Listing[]> {
      const res = await fetch(`${API_URL}/deals`);
      if (!res.ok) {
        throw new Error(`Failed to fetch deals: ${res.status} ${res.statusText}`);
      }
      return res.json();
    }
    
    export function useDeals() {
      return useQuery({
        queryKey: ['deals'],
        queryFn: fetchDeals,
      });
    }
    ```
  - Add to `good-car-deals-frontend/.env.local`: `NEXT_PUBLIC_API_URL=http://localhost:3001`

    // Badge variant mapping for dealType
    function getBadgeVariant(dealType: string | null): 'default' | 'secondary' | 'destructive' | 'outline' {
      if (!dealType) return 'secondary'; // gray/muted for unknown
      const lower = dealType.toLowerCase();
      if (lower.includes('great') || lower.includes('good deal') || lower.includes('decent')) return 'default';
      if (lower.includes('overpriced') || lower.includes('bad') || lower.includes('fair')) return 'destructive';
      return 'outline';
    }
    
    // Format large numbers with commas
    function formatNumber(n: number | null): string {
      if (n === null || n === undefined) return 'N/A';
      return n.toLocaleString('en-US');
    }
    
    export function DealCard({ deal }: { deal: Listing }) {
      return (
        <div className="flex flex-col gap-2 p-4 rounded-lg border bg-card text-card-foreground">
          <div className="flex items-center justify-between gap-4">
            <h3 className="font-heading text-base font-semibold leading-tight line-clamp-2">
              {deal.title}
            </h3>
            <Badge variant={getBadgeVariant(deal.dealType)}>
              {deal.dealType ?? 'Unclassified'}
            </Badge>
          </div>
          
          <Separator />
          
          <div className="grid grid-cols-2 gap-x-4 gap-y-1 text-sm">
            <span className="text-muted-foreground">Price</span>
            <span className="font-medium text-right">{deal.price}</span>
            
            <span className="text-muted-foreground">Model</span>
            <span className="text-right">{deal.model ?? 'N/A'}</span>

            <span className="text-right">{deal.year ?? 'N/A'}</span>

            <span className="text-right">{formatNumber(deal.mileageKm)} km</span>
            
            <span className="text-muted-foreground">Deal Score</span>

              <a href={deal.listingUrl} target="_blank" rel="noopener noreferrer">

          </div>
